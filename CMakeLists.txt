cmake_minimum_required(VERSION 3.16)
project(SPR_Vision_Tutorial_27 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(BUILD_HARDWARE "Build HikRobot and serial hardware drivers" OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d dnn highgui videoio)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(OpenVINO REQUIRED COMPONENTS Runtime HINTS
  /opt/intel/openvino_2024.6.0/runtime/cmake /opt/intel/openvino_2024/runtime/cmake)

add_subdirectory(tools)
add_subdirectory(task/auto_aim)
add_subdirectory(sim_bridge)
if(BUILD_HARDWARE)
  add_subdirectory(io)
  add_executable(task2_real src/task2_real.cpp)
  target_link_libraries(task2_real PRIVATE auto_aim sim_bridge io)
endif()

add_executable(task1_sim src/task1_sim.cpp)
target_link_libraries(task1_sim PRIVATE auto_aim sim_bridge)
