cmake_minimum_required(VERSION 3.16)
project(SPR_Vision_Tutorial_27 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(BUILD_HARDWARE "Build HikRobot and serial hardware drivers" OFF)
option(ENABLE_OPENVINO "Build the YOLOv5 image detector" ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d dnn highgui videoio)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
if(ENABLE_OPENVINO)
  find_package(OpenVINO QUIET COMPONENTS Runtime HINTS
    /opt/intel/openvino_2024.6.0/runtime/cmake /opt/intel/openvino_2024/runtime/cmake)
  if(NOT OpenVINO_FOUND)
    message(WARNING "OpenVINO unavailable: task_sim supports --synthetic and --image-only only")
  endif()
endif()

add_subdirectory(tools)
add_subdirectory(task/auto_aim)
add_subdirectory(sim_bridge)
if(BUILD_HARDWARE)
  add_subdirectory(io)
endif()

add_library(sim_app STATIC src/task1_sim.cpp)
target_link_libraries(sim_app PUBLIC auto_aim sim_bridge)
target_compile_definitions(sim_app PRIVATE PROJECT_SOURCE_PATH="${PROJECT_SOURCE_DIR}")
foreach(app task1_sim task_sim)
  add_executable(${app} src/task_sim.cpp)
  target_link_libraries(${app} PRIVATE sim_app)
endforeach()

include(CTest)
if(BUILD_TESTING)
  add_executable(target_tests tests/target_tests.cpp)
  target_link_libraries(target_tests PRIVATE auto_aim)
  add_test(NAME target_tests COMMAND target_tests)
  add_executable(simulator_tests tests/simulator_tests.cpp)
  target_link_libraries(simulator_tests PRIVATE sim_bridge)
  add_test(NAME simulator_tests COMMAND simulator_tests)
  add_executable(foxglove_tests tests/foxglove_tests.cpp)
  target_link_libraries(foxglove_tests PRIVATE sim_bridge)
  add_test(NAME foxglove_tests COMMAND foxglove_tests)
  add_executable(solver_tests tests/solver_tests.cpp)
  target_link_libraries(solver_tests PRIVATE auto_aim)
  add_test(NAME solver_tests COMMAND solver_tests)
  add_executable(planner_tests tests/planner_tests.cpp)
  target_link_libraries(planner_tests PRIVATE auto_aim)
  add_test(NAME planner_tests COMMAND planner_tests)
  add_executable(shooter_tests tests/shooter_tests.cpp)
  target_link_libraries(shooter_tests PRIVATE auto_aim)
  add_test(NAME shooter_tests COMMAND shooter_tests)
  add_executable(plotter_tests tests/plotter_tests.cpp)
  target_link_libraries(plotter_tests PRIVATE tools)
  add_test(NAME plotter_tests COMMAND plotter_tests)
  set_tests_properties(target_tests simulator_tests foxglove_tests solver_tests planner_tests shooter_tests plotter_tests PROPERTIES TIMEOUT 30)
  add_test(NAME task_sim_smoke COMMAND task_sim --synthetic --duration 1 --no-foxglove)
  set_tests_properties(task_sim_smoke PROPERTIES TIMEOUT 10)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME task_sim_foxglove_integration
      COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/task_sim_integration.py $<TARGET_FILE:task1_sim>)
    set_tests_properties(task_sim_foxglove_integration PROPERTIES TIMEOUT 30)
  endif()
endif()
